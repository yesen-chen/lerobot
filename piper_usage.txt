# ===== wrist 相机视角校准 =====
# 思路：在固定关节位处拍参考图；之后想验证相机角度有没有变，再走到该关节位、把当前帧
# 与参考图做配准（ECC/ORB），输出 dx/dy/rot/scale 和 PASS/WARN/FAIL，并提供 overlay 可视化。
# 注意：record 与 verify 必须使用完全相同的 --robot.cameras 配置（分辨率/旋转/色彩模式），
# 否则脚本会主动拒绝执行。脚本会强制等到关节静止再抓帧。

# 1) 录制参考（首次或重装相机后做一次）：示教模式拖到目标观察位 -> 回车 -> 自动锁位 -> 抓 N 帧中位数 -> 落盘到 outputs/wrist_calib/<name>/
python -m lerobot.scripts.lerobot_wrist_calib record \
  --name=v1 \
  --robot.type=piper --robot.can_name=can0 \
  --robot.cameras='{ wrist: {type: opencv, index_or_path: "/dev/video0", width: 1280, height: 720, fps: 30} }' \
  --camera_key=wrist \
  --avg_frames=10 --settle_eps=0.002 --settle_window=10

# 2) 校验：自动走到记录关节位 -> 实时窗口显示参考/当前/差异三联屏 + HUD（dx/dy/rot/scale/score/PASS-WARN-FAIL）
#    快捷键: q 退出 | s 保存当前帧+视图+指标到 diagnostics/ | e 切 alpha/edge overlay | b 切 ecc/orb backend | m 切 ECC 运动模型
python -m lerobot.scripts.lerobot_wrist_calib verify \
  --name=v1 \
  --robot.type=piper --robot.can_name=can0 \
  --robot.cameras='{ wrist: {type: opencv, index_or_path: "/dev/video0", width: 1920, height: 1080, fps: 30} }' \
  --camera_key=wrist \
  --backend=ecc --ecc_motion=euclidean
